```typescript
/** Payment requirements extracted from a 402 response */
export interface PaymentRequirements {
  scheme: string;           // e.g., "exact"
  network: string;          // v1: "solana-mainnet" | v2: CAIP-2 "solana:5eykt4..."
  maxAmountRequired: string; // amount in smallest unit (e.g., "0" for zero-amount)
  resource: string;         // the URL being accessed
  description?: string;     // human-readable description
  mimeType?: string;        // expected response MIME type
  payTo?: string;           // recipient address
  extra?: Record<string, unknown>; // additional fields
}

/** Payload sent in the payment header */
export interface PaymentPayload {
  x402Version: number;
  scheme: string;
  network: string;
  payload: {
    serializedTransaction: string; // base64-encoded signed Solana tx
  };
}

/** Result from buildPaymentHeader — includes header name and value */
export interface PaymentHeader {
  headerName: string;   // "x-payment" for v1, "payment-signature" for v2
  headerValue: string;  // base64-encoded JSON of PaymentPayload
}
```

Export `PAYMENT_HEADERS` constant:
```typescript
export const PAYMENT_HEADERS = {
  REQUIRED: 'payment-required',
  SIGNATURE: 'payment-signature',
  RESPONSE: 'payment-response',
  V1_PAYMENT: 'x-payment',
  V1_RESPONSE: 'x-payment-response',
} as const;
```

Export `getPaymentVersion(response: Response): 1 | 2 | null`:
- If `response.status !== 402` return `null`
- If `response.headers.get(PAYMENT_HEADERS.REQUIRED)` exists → return `2`
- If `response.headers.get('content-type')?.includes('application/json')` → return `1`
- Otherwise return `null`

Export `async function parsePaymentRequirements(response: Response): Promise<PaymentRequirements[]>`:
- Call `getPaymentVersion(response)` — if `null`, throw `AuthError({ code: 'X402_PARSE_FAILED', message: 'Response is not a valid x402 payment required response' })`
- If version 2: get header, `atob()` decode, `JSON.parse()`, return `.accepts` array. Wrap in try/catch — on error throw `AuthError({ code: 'X402_PARSE_FAILED', message: ... })`
- If version 1: `await response.json()`, return `.accepts` array. Wrap in try/catch — on error throw `AuthError({ code: 'X402_PARSE_FAILED', message: ... })`

Export `function buildPaymentHeader(requirements: PaymentRequirements, signedTx: Uint8Array, version: 1 | 2 = 1): PaymentHeader`:
- Build `PaymentPayload` object with `x402Version: version`, `scheme: requirements.scheme`, `network: requirements.network`, `payload: { serializedTransaction: uint8ArrayToBase64(signedTx) }`
- Encode payload: `btoa(JSON.stringify(paymentPayload))`
- Return `{ headerName: version === 2 ? PAYMENT_HEADERS.SIGNATURE : PAYMENT_HEADERS.V1_PAYMENT, headerValue: encoded }`

Helper (not exported): `function uint8ArrayToBase64(bytes: Uint8Array): string { return btoa(String.fromCharCode(...bytes)); }`

Run tests — they MUST pass.
